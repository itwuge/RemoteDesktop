## 3. 整段可粘贴脚本（含自动 PASS/FAIL）

把下面整块贴进 PowerShell 5.1 直接回车即可；脚本会按 0→18 顺序跑并打印 `[PASS]/[FAIL]`，
结束后打印失败条数。手机需已插线并授权 USB 调试。

```powershell
# ================= RDC 安卓控制端 真机冒烟 A→Z =================
$ADB = "D:\RDC-Build\tools\android\sdk\platform-tools\adb.exe"
$APK = "D:\RDC-Build\artifacts\android\rdc-control-0.1.0-arm64-v8a-release.apk"
$PKG = "com.rdc.remote"
$ACT = "com.rdc.remote/.MainActivity"
$TMP = Join-Path $env:TEMP "rdc-ui.xml"
$script:fail = 0

function Step($name, $ok, $detail) {
  if ($ok) { Write-Host ("[PASS] {0} :: {1}" -f $name, $detail) -ForegroundColor Green }
  else     { Write-Host ("[FAIL] {0} :: {1}" -f $name, $detail) -ForegroundColor Red; $script:fail++ }
}
function Dump-Ui {
  & $ADB shell uiautomator dump /sdcard/rdc-ui.xml | Out-Null
  Start-Sleep -Milliseconds 500
  & $ADB pull /sdcard/rdc-ui.xml $TMP 2>$null | Out-Null
  if (-not (Test-Path $TMP)) { return $null }
  return [xml](Get-Content $TMP -Raw -Encoding UTF8)
}
function Find-Node($xml, $desc) {
  if (-not $xml) { return $null }
  return $xml.SelectNodes("//node") | Where-Object { $_.'content-desc' -eq $desc } | Select-Object -First 1
}
function Find-Text($xml, $text, $clickable) {
  if (-not $xml) { return $null }
  return $xml.SelectNodes("//node") | Where-Object {
    $_.text -eq $text -and ((-not $clickable) -or $_.clickable -eq 'true')
  } | Select-Object -First 1
}
function Tap-Node($node) {
  if (-not $node) { return $false }
  if ($node.bounds -notmatch '\[(\d+),(\d+)\]\[(\d+),(\d+)\]') { return $false }
  $x = [int](([int]$Matches[1] + [int]$Matches[3]) / 2)
  $y = [int](([int]$Matches[2] + [int]$Matches[4]) / 2)
  & $ADB shell input tap $x $y | Out-Null
  Start-Sleep -Milliseconds 600
  return $true
}
function SelectedTab($xml) {
  if (-not $xml) { return "" }
  $n = $xml.SelectNodes("//node") | Where-Object { $_.'content-desc' -like '*已选中*' } | Select-Object -First 1
  if ($n) { return ($n.'content-desc' -replace '，.*$', '') }
  return ""
}

# ---- 0 设备在线
$devs = (& $ADB devices) -join " "
Step "0 设备在线" ($devs -match "\sdevice") $devs.Trim()

# ---- 1 安装
$out = (& $ADB install -r $APK 2>&1 | Out-String)
Step "1 安装" ($out -match "Success") $out.Trim()

# ---- 2 冷启动
& $ADB logcat -c
$out = (& $ADB shell am start -W -n $ACT 2>&1 | Out-String)
Step "2 冷启动" (($out -match "Status: ok") -and ($out -match "COLD")) ($out -replace "`r`n", " ")

# ---- 3 进程与首页日志
Start-Sleep -Seconds 2
$appPid = (& $ADB shell pidof $PKG | Out-String).Trim()
Step "3 进程存活" ($appPid -ne "") "pid=$appPid"
$log = (& $ADB logcat -d -s RdcHome | Out-String)
Step "3b 首页日志" (($log -match "page_ready") -and ($log -match "tabs=4")) "page_ready/tabs=4"

# ---- 4-7 四个 tab 与页面标志文案
$tabs = @(
  @{ Name = "连接"; Mark = "连接到电脑" },
  @{ Name = "设备"; Mark = "同账号设备" },
  @{ Name = "消息"; Mark = "申请与审批" },
  @{ Name = "我的"; Mark = "画质与降级" }
)
$i = 4
foreach ($t in $tabs) {
  $xml = Dump-Ui
  $okTap = Tap-Node (Find-Node $xml $t.Name)
  $xml2 = Dump-Ui
  $sel = SelectedTab $xml2
  $hasMark = $false
  if ($xml2) { $hasMark = ($xml2.OuterXml -like ("*" + $t.Mark + "*")) }
  Step ("{0} tab {1}" -f $i, $t.Name) ($okTap -and ($sel -eq $t.Name) -and $hasMark) "tap=$okTap 选中=$sel 文案=$hasMark"
  $i++
}

# ---- 8 未读红点与全部已读（先切到消息页）
$xml = Dump-Ui
Tap-Node (Find-Node $xml "消息") | Out-Null
$xml = Dump-Ui
Tap-Node (Find-Text $xml "全部已读" $true) | Out-Null
Start-Sleep -Milliseconds 500
$mlog = (& $ADB logcat -d -s RdcMessages | Out-String)
Step "8 全部已读" ($mlog -match "mark_all_read") "mark_all_read"

# ---- 9/10 旋转
$pid1 = (& $ADB shell pidof $PKG | Out-String).Trim()
$pages1 = (([regex]::Matches($log, "page_ready")).Count)
& $ADB shell settings put system accelerometer_rotation 0 | Out-Null
& $ADB shell settings put system user_rotation 1 | Out-Null
Start-Sleep -Seconds 3
$pid2 = (& $ADB shell pidof $PKG | Out-String).Trim()
$log2 = (& $ADB logcat -d -s RdcHome | Out-String)
$pages2 = (([regex]::Matches($log2, "page_ready")).Count)
$sel2 = SelectedTab (Dump-Ui)
Step "9 旋转横屏" (($pid1 -eq $pid2) -and ($pages1 -eq $pages2) -and ($sel2 -ne "")) "pid $pid1->$pid2 page_ready $pages1->$pages2 选中=$sel2"
& $ADB shell settings put system user_rotation 0 | Out-Null
Start-Sleep -Seconds 2
$sel3 = SelectedTab (Dump-Ui)
Step "10 旋转竖屏" ($sel3 -eq $sel2) "选中=$sel3"

# ---- 11 进会话窗（先切回连接页）
$xml = Dump-Ui
Tap-Node (Find-Node $xml "连接") | Out-Null
Start-Sleep -Milliseconds 500
$xml = Dump-Ui
$edits = @()
if ($xml) { $edits = @($xml.SelectNodes("//node[@class='android.widget.EditText']")) }
if ($edits.Count -ge 2) {
  Tap-Node $edits[0] | Out-Null
  & $ADB shell input text "12345678" | Out-Null
  Tap-Node $edits[1] | Out-Null
  & $ADB shell input text "123456" | Out-Null
  $xml = Dump-Ui
  $connBtn = Find-Text $xml "连接" $true
  Tap-Node $connBtn | Out-Null
  Start-Sleep -Seconds 2
  $focus = (& $ADB shell dumpsys window | Select-String "mCurrentFocus" | Out-String)
  Step "11 进会话窗" ($focus -match "SessionActivity") $focus.Trim()
} else {
  Step "11 进会话窗" $false "未找到两个输入框 (count=$($edits.Count))"
}

# ---- 12 会话窗旋转保活
& $ADB shell settings put system user_rotation 1 | Out-Null
Start-Sleep -Seconds 3
$slog = (& $ADB logcat -d -s RdcSession | Out-String)
Step "12 会话窗横屏" (($slog -match "session_kept=true") -or ($slog -match "enter_landscape")) "session_kept/enter_landscape"
& $ADB shell settings put system user_rotation 0 | Out-Null
Start-Sleep -Seconds 2

# ---- 13/14 触摸与按键打到命令入口
$xml = Dump-Ui
$surf = Find-Node $xml "远端画面区域, 可触摸操作"
Tap-Node $surf | Out-Null
& $ADB shell input keyevent 66 | Out-Null
Start-Sleep -Milliseconds 800
$clog = (& $ADB logcat -d -s RdcContract | Out-String)
$injects = ([regex]::Matches($clog, "\[inject\]")).Count
Step "13/14 输入打到契约" ($injects -ge 2) "inject 条数=$injects"

# ---- 15 返回保持 tab
& $ADB logcat -c
& $ADB shell input keyevent 4 | Out-Null
Start-Sleep -Seconds 2
$selBack = SelectedTab (Dump-Ui)
Step "15 返回保持 tab" ($selBack -eq "连接") "选中=$selBack"

# ---- 16 退出
& $ADB shell input keyevent 4 | Out-Null
Start-Sleep -Seconds 2
$focus2 = (& $ADB shell dumpsys window | Select-String "mCurrentFocus" | Out-String)
Step "16 退出" ($focus2 -notmatch $PKG) $focus2.Trim()

# ---- 17 崩溃检查
$crash = (& $ADB logcat -d -b crash | Out-String)
Step "17 无崩溃" ($crash -notmatch $PKG) "crash 记录"

# ---- 18 卸载
$out = (& $ADB uninstall $PKG 2>&1 | Out-String)
Step "18 卸载" ($out -match "Success") $out.Trim()

& $ADB shell settings put system accelerometer_rotation 1 | Out-Null
Write-Host ("==== 失败条数: {0} ====" -f $script:fail) -ForegroundColor Yellow
```

## 5. 脚本已知局限

1. 脚本未在真机验证过；如某步因 UI 层级变化失败，按第 2 节表格手动执行同一条 `adb` 命令即可。
2. tab 与按钮靠 `uiautomator dump` 的 `content-desc` / `text` 定位，依赖本版界面文案；文案改动后需同步第 4 节脚本里的 `$tabs` 与按钮文本。
3. 会话窗仍为离线预览（`ControlContract.WIRED=false`），第 13/14 步只证明「触摸/按键到达契约入口」，**不代表真实远控可用**。
4. 截图取证建议在 4/5/6/7/12 步各存一张：`adb exec-out screencap -p > evidence\真机-步骤N.png`。
